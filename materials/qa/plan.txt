Примеры ревью:

@Service

public class OrderService {



    @Autowired

    private OrderRepository orderRepository;



    private RestTemplate restClient = MyRestTemplate();



    @Transactional

    public void createOrder(Order order) {

        orderRepository.save(order);

        String paymentStatus = restClient.postForObject(

                "https://api.payments.com", order, String.class);

        System.out.println("request is done");

        if ("SUCCESS".equals(paymentStatus)) {

            order.setStatus("PAID");

            orderRepository.save(order);

        } else {

            throw new RuntimeException("Payment failed");
        }
    }
}



@Service
public class OrderService {

    @Autowired
    private OrderRepository orderRepository;

    @Autowired
    private PaymentService paymentService; // Класс без @Transactional

    public void createOrder(Order order) {
        // 1. Транзакция только на работу с базой
        saveOrderInternal(order);
        // 2. Внешний вызов БЕЗ активной транзакции
        paymentService.charge(order);
    }


    @Transactional
    public void saveOrderInternal(Order order) {
        orderRepository.save(order);
    }
}
